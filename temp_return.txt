  return (
    <div className="absolute left-0 right-0 top-[64px] md:top-[80px] bottom-[50px] flex flex-col overflow-hidden bg-transparent text-gray-900 dark:text-gray-100 font-sans transition-colors z-10" onClick={() => { setIsScriptMenuOpen(false); setIsModelMenuOpen(false); }}>
      
      <main className="flex-1 overflow-y-auto w-full flex flex-col items-center">
        {chatHistory.length === 0 ? (
          <div className="w-full max-w-3xl flex-1 flex flex-col items-center justify-center px-4 -mt-10">
            <h2 className="text-3xl md:text-4xl font-black font-heading tracking-tight mb-4 text-gray-900 dark:text-white flex items-center gap-3">
              Ebook Prompt Builder <Book className="text-rn-accent inline-block ml-2" size={32} />
            </h2>
            <p className="text-center text-gray-500 dark:text-gray-400 mb-8 max-w-lg text-sm">
              Rancang kerangka dan hasilkan master prompt terstruktur untuk ebook premium Anda secara otomatis.
            </p>

            <div className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-white/10 rounded-2xl shadow-sm flex flex-col focus-within:ring-2 focus-within:ring-rn-accent/50 focus-within:border-rn-accent transition-all duration-300 relative z-10">
              <textarea 
                ref={textareaRef}
                value={ebookPromptText + (interimText ? (ebookPromptText && !ebookPromptText.endsWith(' ') ? ' ' : '') + interimText : '')}
                onChange={autoResizeTextarea}
                onKeyDown={(e) => {
                  if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    generateEbookPrompt(null);
                  }
                }}
                className="w-full bg-transparent p-4 min-h-[140px] max-h-[250px] resize-none outline-none text-gray-900 dark:text-white placeholder-gray-400 text-sm md:text-base leading-relaxed"
                placeholder="Topik: Panduan Memulai Bisnis Digital tanpa Modal..."
              />
              
              <div className="px-4 pb-3 pt-1 flex items-center justify-between border-t border-transparent bg-transparent relative z-20">
                {renderInputControls()}
                
                <div className="flex items-center gap-1.5 md:gap-2 shrink-0">
                  <button 
                    onClick={() => {}} 
                    className="w-8 h-8 md:w-9 md:h-9 rounded-full flex items-center justify-center shrink-0 transition-colors border bg-transparent text-gray-400 hover:bg-gray-50 dark:hover:bg-white/5 border-transparent"
                    title="Input menggunakan suara (Segera Hadir)"
                  >
                    <Mic size={18} />
                  </button>
                  <button 
                    disabled={isLoading}
                    onClick={() => generateEbookPrompt(null)}
                    className="w-8 h-8 md:w-9 md:h-9 rounded-full bg-rn-accent text-white flex items-center justify-center shrink-0 disabled:opacity-50 disabled:cursor-not-allowed hover:bg-orange-600 transition-colors shadow-sm"
                  >
                    {isLoading ? <Square size={14} className="fill-current" strokeWidth={2.5} /> : <Send size={14} className="-ml-0.5" strokeWidth={2.5}/>}
                  </button>
                </div>
              </div>
            </div>
          </div>
      ) : (
        <div className="w-full max-w-3xl mx-auto px-4 flex flex-col space-y-8 pb-8 pt-6">
            {chatHistory.map((msg, idx) => (
              <div key={msg.id} className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'} animate-in fade-in slide-in-from-bottom-2 duration-300`}>
                
                {msg.role === "user" && (
                  <div className="flex flex-col items-end w-full">
                    <div className="bg-gray-100 dark:bg-white/10 text-gray-800 dark:text-gray-200 px-5 py-4 rounded-[20px] rounded-tr-sm max-w-[85%] md:max-w-[75%] shadow-sm border border-gray-200 dark:border-white/5">
                      <p className="text-[14px] whitespace-pre-wrap leading-relaxed font-medium">{msg.content}</p>
                    </div>
                  </div>
                )}
                
                {msg.role === "assistant" && (
                  <div className="flex flex-col w-full">
                    {msg.data?.is_error ? (
                      <div className="flex justify-start">
                        <div className="bg-red-50 dark:bg-red-900/20 text-red-600 dark:text-red-400 px-5 py-4 rounded-[20px] rounded-tl-sm border border-red-200 dark:border-red-900/30 inline-block break-words whitespace-pre-wrap max-w-[85%] md:max-w-[75%]">
                          <p className="text-[14px]">{msg.data.error_message}</p>
                        </div>
                      </div>
                    ) : (
                      <div className="space-y-6 w-full mt-2">
                        {/* Message from AI */}
                        {msg.data?.message && (
                          <div className="mb-4 text-sm text-gray-700 dark:text-gray-300 prose dark:prose-invert prose-p:leading-relaxed prose-headings:font-heading prose-headings:font-bold prose-headings:mb-2 prose-headings:mt-4 prose-li:my-0.5 max-w-none">
                            <ReactMarkdown remarkPlugins={[remarkGfm]}>{msg.data.message}</ReactMarkdown>
                          </div>
                        )}

                        {/* Ideas Table (Brainstorming Mode) */}
                        {msg.data?.ideas_table && msg.data.ideas_table.length > 0 && (
                          <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mt-4">
                            {msg.data.ideas_table.map((idea, i) => (
                              <div key={i} className="bg-white dark:bg-[#121214] border border-gray-200 dark:border-white/10 rounded-2xl overflow-hidden shadow-sm hover:shadow-md transition-all duration-300 flex flex-col relative">
                                <div className="p-4 border-b border-gray-200 dark:border-white/10 bg-gray-50/50 dark:bg-white/5">
                                  <div className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md bg-rn-accent/10 text-rn-accent text-[10px] font-bold uppercase tracking-wider mb-2 font-heading border border-rn-accent/20">
                                    <Lightbulb size={12} /> Ide {i + 1}
                                  </div>
                                  <h4 className="font-bold text-gray-900 dark:text-white font-heading text-lg leading-tight">{idea.topic || idea.nama_ide}</h4>
                                </div>
                                <div className="p-4 flex-1 space-y-3">
                                  <div>
                                    <span className="text-[10px] font-bold text-gray-500 uppercase">Masalah Audiens</span>
                                    <p className="text-xs text-gray-800 dark:text-gray-300">{idea.masalah_audiens}</p>
                                  </div>
                                  <div>
                                    <span className="text-[10px] font-bold text-gray-500 uppercase">Unique Angle</span>
                                    <p className="text-xs text-gray-800 dark:text-gray-300">{idea.unique_angle || idea.angle}</p>
                                  </div>
                                  <div>
                                    <span className="text-[10px] font-bold text-gray-500 uppercase">Dream Outcome</span>
                                    <p className="text-xs text-rn-accent font-semibold">{idea.dream_outcome}</p>
                                  </div>
                                </div>
                                <div className="p-3 border-t border-gray-200 dark:border-white/10 bg-gray-50/50 dark:bg-white/5">
                                  <button onClick={() => {
                                      const autoPrompt = `[SYSTEM_OUTLINE] Saya tertarik dengan ide ini:\nTopik: ${idea.topic || idea.nama_ide}\nAngle: ${idea.unique_angle || idea.angle}\n\nTolong buatkan saya kerangka Outline (Daftar Isi) beserta Bab, Sub-bab, dan Inti Pembahasan yang sangat detail berdasarkan referensi gaya ebook premium.`;
                                      generateEbookPrompt(autoPrompt, true);
                                  }} className="w-full flex items-center justify-center gap-2 py-2 px-3 rounded-lg bg-rn-accent text-white text-xs font-bold hover:bg-orange-600 transition-colors shadow-sm">
                                    <Book size={14} /> Pilih Ide Ini
                                  </button>
                                </div>
                              </div>
                            ))}
                          </div>
                        )}

                        {/* REVIEW OUTLINE MODE */}
                        {msg.type === 'review_outline' && msg.data && (
                          <div className="space-y-4 mt-4 w-full">
                            {msg.isReviewing ? (
                               <div className="bg-white dark:bg-[#121214] border border-gray-200 dark:border-white/20 rounded-2xl p-5 shadow-sm">
                                  <div className="bg-amber-50 dark:bg-amber-900/20 text-amber-800 dark:text-amber-400 px-4 py-3 rounded-xl text-xs md:text-sm font-bold flex items-start gap-2 border border-amber-200 dark:border-amber-800/30 mb-4">
                                      <AlertCircle size={18} className="shrink-0 mt-0.5" />
                                      <span>Review & Edit Outline ini. Jika sudah mantap, klik "Generate Master Prompt". AI akan memecah instruksi ini menjadi blok prompt untuk ChatGPT.</span>
                                  </div>
                                  <textarea
                                      value={msg.data.outlineText || ''}
                                      onChange={(e) => setChatHistory(prev => prev.map(m => m.id === msg.id ? { ...m, data: { ...m.data, outlineText: e.target.value } } : m))}
                                      className="w-full h-[350px] p-4 text-sm leading-relaxed rounded-xl border border-gray-200 dark:border-white/10 bg-gray-50 dark:bg-white/5 text-gray-900 dark:text-gray-200 outline-none resize-y focus:ring-2 focus:ring-rn-accent"
                                  />
                                  <div className="flex justify-end pt-4 mt-4 border-t border-gray-200 dark:border-white/10">
                                      <button
                                          onClick={() => {
                                              setChatHistory(prev => prev.map(m => m.id === msg.id ? { ...m, isReviewing: false } : m));
                                              const finalPrompt = `[SYSTEM AUTO-SUBMIT] Saya telah menyetujui Outline ini. Tolong jalankan MODE C (Prompt Production) dan cetak Master Prompt berupa block-block terpisah untuk diberikan ke ChatGPT/Claude berdasarkan Outline final berikut:\n\n${msg.data?.outlineText}`;
                                              generateEbookPrompt(finalPrompt);
                                          }}
                                          className="flex items-center gap-2 px-6 py-2.5 rounded-xl font-bold bg-rn-accent text-white hover:bg-orange-600 transition-all shadow-sm text-sm"
                                      >
                                          <Send size={16} strokeWidth={2.5}/> Generate Master Prompt
                                      </button>
                                  </div>
                               </div>
                            ) : (
                               <div className="bg-emerald-50 dark:bg-emerald-900/20 text-emerald-700 dark:text-emerald-400 px-4 py-3 rounded-xl text-sm font-bold flex items-center gap-2 border border-emerald-200 dark:border-emerald-800/30">
                                  <CheckCircle2 size={18} className="shrink-0" /> Outline Ebook telah dikunci dan dikirim untuk diproses menjadi Prompt.
                               </div>
                            )}
                          </div>
                        )}

                        {/* CP Result (Production Mode) */}
                        {msg.type === 'ebook_result' && msg.data && (
                          <div className="space-y-4 mt-4 w-full">
                            <h4 className="font-bold text-gray-900 dark:text-white font-heading flex items-center gap-2 text-sm md:text-base mb-3 mt-6">
                              <CheckCircle2 size={18} className="text-rn-accent" strokeWidth={2.5}/> 
                              Master Prompt Ebook ({msg.data.prompt_blocks?.length || 0} Blok)
                            </h4>
                            <div className="bg-blue-50 dark:bg-blue-900/20 text-blue-800 dark:text-blue-400 px-4 py-3 rounded-xl text-xs md:text-sm font-semibold flex items-start gap-2 border border-blue-200 dark:border-blue-800/30 mb-4">
                              <p>💡 <b>Cara Pakai:</b> Copy prompt berbahasa Indonesia di bawah ini SECARA BERURUTAN (mulai dari blok 1) dan berikan ke ChatGPT atau Claude Anda. AI tersebut akan langsung merespons dengan konten Ebook yang sangat relevan dan mendalam sesuai strategi Anda!</p>
                            </div>
                            
                            {msg.data.prompt_blocks && msg.data.prompt_blocks.map((block, i) => (
                              <div key={i} className="bg-[#0D0D0D] border border-gray-800 rounded-xl overflow-hidden shadow-lg relative group">
                                <div className="bg-gray-900 border-b border-gray-800 px-4 py-2.5 flex items-center justify-between">
                                  <div className="flex items-center gap-2">
                                    <div className="w-2.5 h-2.5 rounded-full bg-rn-accent"></div>
                                    <span className="text-xs font-bold text-gray-300 font-heading tracking-wider uppercase">{block.label}</span>
                                  </div>
                                  <button onClick={() => navigator.clipboard.writeText(block.content)} className="text-gray-400 hover:text-white transition-colors p-1" title="Salin Prompt">
                                    <Copy size={14} />
                                  </button>
                                </div>
                                <div className="p-4 text-[13px] md:text-[14px] text-gray-300 font-mono whitespace-pre-wrap leading-relaxed overflow-x-auto">
                                  {block.content}
                                </div>
                              </div>
                            ))}
                          </div>
                        )}
                      </div>
                    )}
                  </div>
                )}
              </div>
            ))}
            
            {isLoading && (
              <div className="flex justify-start">
                <div className="flex gap-1.5 p-4 bg-gray-50 dark:bg-[#121214] border border-gray-100 dark:border-white/5 rounded-2xl rounded-tl-sm w-fit">
                  <span className="w-2 h-2 rounded-full bg-gray-400 dark:bg-gray-500 animate-bounce"></span>
                  <span className="w-2 h-2 rounded-full bg-gray-400 dark:bg-gray-500 animate-bounce [animation-delay:0.2s]"></span>
                  <span className="w-2 h-2 rounded-full bg-gray-400 dark:bg-gray-500 animate-bounce [animation-delay:0.4s]"></span>
                </div>
              </div>
            )}
            <div ref={chatEndRef} className="h-4" />
          </div>
      )}
      </main>

      {chatHistory.length > 0 && (
        <div className="w-full shrink-0 bg-white/90 dark:bg-rn-dark-bg/90 backdrop-blur-md pt-2 pb-4 px-4 z-40">
          <div className="max-w-3xl mx-auto relative">
            <div className="w-full bg-white dark:bg-[#121214] border border-gray-200 dark:border-white/10 rounded-2xl shadow-lg flex flex-col focus-within:ring-2 focus-within:ring-rn-accent/50 focus-within:border-rn-accent transition-all duration-300 relative z-10">
              <textarea 
                value={ebookPromptText + (interimText ? (ebookPromptText && !ebookPromptText.endsWith(' ') ? ' ' : '') + interimText : '')}
                onChange={autoResizeTextarea}
                onKeyDown={(e) => {
                  if (e.key === 'Enter' && !e.shiftKey) {
                    e.preventDefault();
                    generateEbookPrompt(null);
                  }
                }}
                className="w-full bg-transparent px-4 py-3 min-h-[48px] max-h-[150px] resize-none outline-none text-gray-900 dark:text-white placeholder-gray-400 text-sm leading-relaxed"
                placeholder="Ketik balasanmu..."
              />
              
              <div className="px-4 pb-2 pt-0 flex items-center justify-between border-t border-transparent bg-transparent relative z-20">
                {renderInputControls()}
                
                <div className="flex items-center gap-1.5 md:gap-2 shrink-0">
                  <button 
                    onClick={() => {}} 
                    className="w-8 h-8 md:w-9 md:h-9 rounded-full flex items-center justify-center shrink-0 transition-colors border bg-transparent text-gray-400 hover:bg-gray-50 dark:hover:bg-white/5 border-transparent"
                    title="Input menggunakan suara (Segera Hadir)"
                  >
                    <Mic size={18} />
                  </button>
                  <button 
                    disabled={isLoading || (!ebookPromptText.trim() && !interimText)}
                    onClick={() => generateEbookPrompt(null)}
                    className="w-8 h-8 md:w-9 md:h-9 rounded-full bg-rn-accent text-white flex items-center justify-center shrink-0 disabled:opacity-50 disabled:cursor-not-allowed hover:bg-orange-600 transition-colors shadow-sm"
                  >
                    {isLoading ? <Square size={14} className="fill-current" strokeWidth={2.5} /> : <Send size={14} className="-ml-0.5" strokeWidth={2.5}/>}
                  </button>
                </div>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
